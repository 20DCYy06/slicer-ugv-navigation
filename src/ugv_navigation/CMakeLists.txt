cmake_minimum_required(VERSION 3.0.2)
project(ugv_navigation)
find_package(catkin REQUIRED)
catkin_package()
catkin_install_python(PROGRAMS scripts/goal.py scripts/depth_obstacles.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(FILES scripts/depth_geometry.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  catkin_add_nosetests(test/test_goal_client.py)
  catkin_add_nosetests(test/test_depth_geometry.py)
  add_rostest(test/odom.test)
  add_rostest(test/client.test)
  add_rostest(test/guarded.test)
endif()
